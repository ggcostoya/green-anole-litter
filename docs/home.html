micro_df <- readRDS("data/micro_df.rds")

# plot temperature maps
micro_df |>
  filter(as.character(z) %in% c("-0.04", "1.8")) |>
  mutate(
    z = factor(
      z,
      levels = c("1.8", "-0.04"),
      labels = c(
        "Lizard Body Temperature in <span style='color: forestgreen;'><b>Canopy</b></span>",
        "Lizard Body Temperature in <span style='color: darkorange;'><b>Leaf Litter</b></span>"
      )
    )
  ) |>
  ggplot(aes(x = lon, y = lat)) +
  geom_tile(aes(fill = min_annual_body_temp)) +
  geom_tile(
    data = micro_df |>
      select(lon, lat, presence) |>
      distinct() |> 
      filter(presence == 1),
    aes(x = lon, y = lat),
    fill = "forestgreen",
    alpha = 0.2
  ) +
  geom_contour(
    aes(x = lon, y = lat, z = presence, color = "Range"),
    linewidth = 0.5
  ) +
  geom_contour(
    aes(x = lon, y = lat, z = min_annual_body_temp, color = "0 °C"),
    breaks = c(0),
    linewidth = 1
  ) +
  geom_contour(
    aes(x = lon, y = lat, z = min_annual_body_temp, color = "CTmin"),
    breaks = c(7),
    linewidth = 1
  ) +
  scale_color_manual(
    name = NULL,
    values = c(
      "0 °C" = "steelblue4",
      "CTmin" = "skyblue",

    labels = expression(bold("0 °C"), bold(CT[min]), bold("Range"))

  scale_fill_viridis(
    option = "magma",
    guide = guide_colorbar(
      title.position = "top",
      label.position = "left",
      barheight = unit(5, "cm")

  ) +
  scale_x_continuous(
    breaks = seq(-100, -70, 5),
    labels = paste0(seq(100, 70, -5), "°W"),
    expand = c(0, 0)
  ) +
  scale_y_continuous(
    breaks = seq(0, 45, 2.5),
    labels = paste0(seq(0, 45, 2.5), "°N"),
    expand = c(0, 0)
  ) +
  facet_wrap(
    ~ z,
    ncol = 1,
    axes = "all"
  ) +
  coord_quickmap(xlim = c(-101, -75), ylim = c(24, 38)) +
  theme_minimal() +
  guides(
    color = guide_legend(
      label.position = "left",

        legend.text = element_text(size = 18, face = "bold"),

# read data
otm_data <- readRDS("data/otms-data.rds")

# plot example OTM dynamic
otm_data |>
  filter(location %in% c("ATL", "MIA")) |>
  mutate(
    location = factor(
      location,
      levels = c("MIA", "ATL"),
      labels = c("Miami (25.7 °N)", "Atlanta (33.7 °N)")
    )
  ) |>
  filter(doy == 19) |>
  mutate(hour = round(mod / 60)) |>
  group_by(location, microhabitat, hour) |>
  summarise(
    mean_temp = mean(temp, na.rm = TRUE),
    sd_temp = sd(temp, na.rm = TRUE),
  ) |>
  ggplot(aes(x = hour, y = mean_temp, color = microhabitat)) +
  annotate(
    "segment",
    x = -Inf,
    xend = Inf,
    y = 0,
    yend = 0,
    linetype = 2,
    color = "steelblue4"
  ) +
  annotate(
    "text",
    x = 18,
    y = 9.5,
    label = expression(bold(CT[min])),
    color = "skyblue",
    hjust = 0,
    vjust = 0,
    size = 5
  ) +
  annotate(
    "segment",
    x = -Inf,
    xend = Inf,
    y = 9,
    yend = 9,
    linetype = 2,
    color = "skyblue"
  ) +
  geom_ribbon(
    aes(
      ymin = mean_temp - sd_temp,
      ymax = mean_temp + sd_temp,
      fill = microhabitat
    ),
    alpha = 0.2
  ) +
  scale_fill_manual(
    values = c("forestgreen", "darkorange"),
    labels = c("Canopy", "Leaf Litter"),

  scale_color_manual(
    values = c("forestgreen", "darkorange"),
