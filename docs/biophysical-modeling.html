caro_occ_data <- read.csv("raw_data/anolis_carolinensis_inat_obs.csv")

# round coordinates and remove observations with low positional accuracy

  mutate(
    lat = round(latitude, 1), lon = round(longitude, 1),
  ) |>
  filter(
    !is.na(positional_accuracy) & positional_accuracy <= 1000
  ) |>
  group_by(lon, lat) |>
  summarise(n = n(), .groups = "drop") |>
  filter(n > 2) |>
  select(lon, lat)

# filter data to native range of green anole
caro_occ_data <- caro_occ_data |>
  filter(
    lon > -110 & lon < -70, lat > 22.5 & lat < 37.5
  )

# remove outliers based on distance to other observations
caro_occ_data$is_outlier <- apply(
  caro_occ_data[, c("lon", "lat")], 1, function(x) {
    distances <- geosphere::distm(
      x,
      caro_occ_data[, c("lon", "lat")],
      fun = distHaversine
    )
    # if there are fewer than 3 observations within 50 km, mark as outlier
    if (length(distances) > 0 && sum(distances < 50000) < 3) {
      1
    } else {
      0
    }
  }
)

# save processed occurrence data
saveRDS(caro_occ_data, "data/caro_occ_data.rds")

# transform into a spatial object
caro_occ_sf <- st_as_sf(
  caro_occ_data[caro_occ_data$is_outlier == 0, ],
  coords = c("lon", "lat"),
  crs = 4326
)

# manually draw map around range
mapviewOptions(fgb = FALSE)
caro_range_sf <- editMap(mapview(caro_occ_sf))

# transform into a polygon object
caro_range_sf <- st_as_sf(caro_range_sf$finished)

# filter data to only include points within the continental US
caro_range_sf <- st_intersection(
  caro_range_sf,
  ne_countries(scale = "medium", returnclass = "sf") |>
    filter(admin %in% c("United States of America", "Mexico")) |>
    st_crop(xmin = -125, xmax = -65, ymin = 20, ymax = 50)
)

# change the spatial resolution of the polygon to 0.04166667 degrees (~5 km)
caro_range_sf <- st_transform(caro_range_sf, crs = 4326)

# combine into a single (multi)polygon suitable for masking other layers
caro_range_poly <- st_as_sf(caro_range_sf) |>

In the plot below, we visualize how the green anole’s range is distributed across different temperature thresholds imposed by the physiology and behavior of the species. The thresholds are defined as follows:

  • Hard Threshold: Minimum annual temperature below 0°C, which is lethal for the species.
  • Soft Threshold: Minimum annual temperature between 0°C and 5.35°C, which is the lowest ever empirically estimated CTmin.
  • Behavioral Threshold: Minimum annual temperature between 5.35°C and 6.6°C, which is the temperature at which there is < 10% of observing green anoles perching at night based on our surveys.

The green anole’s cold-range overfill

A key question is the extent to which the green anole’s range is overfilled on the cold side, meaning what % of its range experiences a minimum annual temperature below it’s CTmin. # define physical constants
k <- 0.05 # thermal conductivity of leaf litter (W m^-1 K^-1)
rho <- 150 # bulk density of leaf litter (kg m^-3)
cp <- 2500 # specific heat capacity of leaf litter (J kg^-1 K^-1)
cc <- 0.007 # canopy cooling coefficient (W m^-2)
srh <- 0.02 # surface radiative heat flux coefficient (C / W m^-2 K^-1)
ecc <- 0.0167 # earth orbit eccentricity
spo <- 1.36 # extraterrestial solar radiation (kW m^-2)
atc <- 0.83 # atmospheric transmissivity coefficient (unitless)

# calculate derived physical constants
diff_sec <- k / (rho * cp) # thermal diffusivity of leaf litter (m^2 s^-1)
diff_hour <- diff_sec * 3600 # thermal diffusivity of leaf litter (m^2 h^-1)
omega <- 2 * pi / 24 # angular frequency of diurnal temperature cycle (rad h^-1)
damping_depth <- sqrt((2 * diff_hour) / omega) # damping depth (m)

# define list of heights and depths of interest
zs <- c(seq(-0.2, -0.01, by = 0.01), 0, 1.8, 2.2, 3)

# define function to calculate solar radiation profile
get_solar_rad_profile <- function(lat, elev, doy, solar_noon) {
  hours <- 0:23
  elon <- (2 * pi / 365) * (doy - 80) + 2 * ecc * (sin(2 * pi / 365 * doy) - sin(2 * pi / 365 * 80)) # nolint
  dec_ang <- asin(0.3879672 * sin(elon))
  p_a <- 101.3 * exp(-elev / 8200)
  geo_lat <- lat * (pi / 180)
  sin_alt_ang <- sin(geo_lat) * sin(dec_ang) + cos(geo_lat) * cos(dec_ang) * outer(solar_noon, hours, function(s, h) cos(0.2618 * (h - s))) # nolint
  m_a <- p_a / (101.3 * (1 / sin_alt_ang))
  solar_rad <- spo * atc ^ m_a * sin_alt_ang
  solar_rad[solar_rad < 0] <- 0
  drop(solar_rad * 1000)
}

# define function to calculate air temperature profile
get_air_temp_profile <- function(tmax, tmin) {
  w <- pi / 12
  gamma <- 0.44 - 0.46 * sin(0.9 + w * (0:23)) + 0.11 * sin(0.9 + 2 * w * (0:23)) # nolint
  drop(outer(tmax, gamma) + outer(tmin, 1 - gamma))
}

# define function to correct air temperature profile from canopy cooling
correct_air_temp_profile <- function(
    air_temp_profile, cc = 0.007, solar_rad_profile, veg_cover) {
  is_vec <- is.vector(air_temp_profile)
  if (is_vec) {
    air_temp_profile  <- matrix(air_temp_profile,  nrow = 1)
    solar_rad_profile <- matrix(solar_rad_profile, nrow = 1)
  }
  result <- air_temp_profile - sweep(cc * solar_rad_profile, 1, veg_cover, "*")
  if (is_vec) drop(result) else result
}

# define function to calculate surface temperature profile
get_surf_temp_profile <- function(
    air_temp_profile, solar_rad_profile, srh = 0.02, veg_cover) {
  is_vec <- is.vector(air_temp_profile)
  if (is_vec) {
    air_temp_profile  <- matrix(air_temp_profile,  nrow = 1)
    solar_rad_profile <- matrix(solar_rad_profile, nrow = 1)
  }
  result <- air_temp_profile + sweep(srh * solar_rad_profile, 1, 1 - veg_cover, "*") # nolint
  if (is_vec) drop(result) else result
}

get_z_temp_profile <- function(
  z_list, surf_temp_profile, air_temp_profile, damping_depth, ref_height = 1.8
) {

  # check if the input profiles are vectors or matrices
  is_vec <- is.vector(surf_temp_profile)
  if (is_vec) {
    surf_temp_profile <- matrix(surf_temp_profile, nrow = 1)
    air_temp_profile  <- matrix(air_temp_profile,  nrow = 1)
  }

  # define hours and extended hours for interpolation
  hours <- 0:23
  surf_temp_mean <- rowMeans(surf_temp_profile)
  extended_hours <- c(hours - 24, hours, hours + 24)

  # calculate temperature profiles at different heights and depths
  z_temp_profiles <- lapply(z_list, function(z) {
    if (z == 0) {
      surf_temp_profile
    } else if (z == ref_height) {
      air_temp_profile
    } else if (z > 0) {
      a <- log((z + 0.01) / 0.01) / log((ref_height + 0.01) / 0.01)
      surf_temp_profile - (surf_temp_profile - air_temp_profile) * a
    } else {
      damp_ratio <- exp(-abs(z) / damping_depth)
      phase_lag <- (abs(z) / damping_depth) * (24 / (2 * pi))
      xout <- hours - phase_lag
      idx <- findInterval(xout, extended_hours)
      w <- (xout - extended_hours[idx]) / (extended_hours[idx + 1] - extended_hours[idx]) # nolint
      ext_surf <- cbind(surf_temp_profile, surf_temp_profile, surf_temp_profile)
      shifted_temp <- sweep(ext_surf[, idx],     2, 1 - w, "*") + sweep(ext_surf[, idx + 1], 2, w, "*") # nolint
      surf_temp_mean + (shifted_temp - surf_temp_mean) * damp_ratio
    }
  })

  # combine the temperature profiles into a matrix
  names(z_temp_profiles) <- z_list
  if (is_vec) z_temp_profiles <- lapply(z_temp_profiles, drop)
  z_temp_profiles

}

# summarize zs temperature profiles into a data frame
microclimate_temps <- Map(
  function(z_temp, z_val) {
    data.frame(
      lon = eco_df$lon,
      lat = eco_df$lat,
      elev = eco_df$elevation,
      veg = eco_df$veg,
      presence = eco_df$present,
      z  = z_val,
      min_temp = do.call(pmin, as.data.frame(z_temp)),
      hours_below_0C   = rowSums(z_temp < 0),
      hours_below_6.8C = rowSums(z_temp < 6.8)
    )
  },
  zs_temp_profiles, zs
) |>
  bind_rows() |>
  group_by(lon, lat, elev, veg, presence, z) |>
  summarise(
    min_annual_temp   = min(min_temp),
    hours_below_0C    = sum(hours_below_0C),
    hours_below_ctmin = sum(hours_below_6.8C),
    .groups = "drop"
  )

# Function to transform minimum annual temperatures into body temperatures
get_body_temp <- function(ambient_t, mass, in_litter) {

  # Define physical constants
  sigma <- 5.670374e-8       # Stefan-Boltzmann constant (W m^-2 K^-4)
  lizard_emissivity <- 0.95  # Longwave emissivity of reptile skin
  sky_emissivity <- 0.75     # Longwave emissivity of the night sky
  wind_speed <- 0.1          # Wind speed in m/s
  f_sky <- 0.5               # Sky view factor above ground (50%)
  f_ground <- 0.5            # Ground view factor above ground (50%)
  convection_ground <- 2.5   # Conductive/convective transfer from ground

  # Convert ambient temp to Kelvin
  ambient_t_k <- ground_t_k <- ambient_t + 273.15

  # Characteristic length of lizard (meters), mass converted from grams to kg
  l <- 0.02 * (mass / 1000)^(1 / 3)

  # Heat transfer conductances

  # Forced convection coefficient
  forced_convection <- 10.3 * (wind_speed^0.6) / (l^0.4)

  # Free convection baseline
  free_convection <- 3.5 * (1 / l)^0.25

  # Total convective heat transfer coefficient
  total_convection <- pmax(forced_convection, free_convection)

  # Radiative heat transfer coefficient
  rad_heat_transfer <- 4 * lizard_emissivity * sigma * (ambient_t_k^3)

  # Night sky radiative temperature (Kelvin)
  sky_t_k <- (sky_emissivity)^0.25 * ambient_t_k

  # Equivalent radiant temperature (°C)
  radiant_t_k <- (f_sky * (sky_t_k^4) + f_ground * (ground_t_k^4))^0.25
  radiant_t <- radiant_t_k - 273.15

  # Operative temperature equilibrium above ground (°C)
  a <- (
    total_convection * ambient_t +
    convection_ground * ambient_t +
    rad_heat_transfer * radiant_t
  )
  b <- total_convection + convection_ground + rad_heat_transfer
  
  # Weighted average (division, not addition)
  op_t <- a / b

  ifelse(in_litter, ambient_t, op_t)

}
