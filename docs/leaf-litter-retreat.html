otms_data <- readRDS("data/otms-data.rds")

# latitudes of locations
otms_data <- otms_data |>
  mutate(
    lat = case_when(
      location == "MIA" ~ 25.7617,
      location == "PSL" ~ 27.305,
      location == "ORL" ~ 28.5383,
      location == "GNV" ~ 29.6516,
      location == "ATL" ~ 33.7490
    )
  )

# predict CTmin by latitude
pred_ctmin <- tibble(
  lat = seq(25, 35, by = 0.1),
  ctmin = 57 - 1.68 * lat - 41.67 + 1.48 * lat
)

# read and summarize survey data
survey <- readRDS("data/sleepy-lizard-survey-data.rds")
survey_sum <- survey |>
  group_by(year, doy) |>
  summarise(
    count = n() - 1,
    presence = ifelse(count > 0, 1, 0)
  ) |>
  mutate(
    continuous_doy = ifelse(year == 2026, doy + 365, doy)
  ) |>
  filter(continuous_doy < (365 + 30)) |>
  mutate(
    date = as.Date(paste(year, doy), format = "%Y %j")
  ) |>
  ungroup()

# estimate sunset times for Atlanta, GA
my_year <- 2026
my_dates <- seq(
  as.Date(sprintf("%d-01-01", my_year)),
  as.Date(sprintf("%d-12-31", my_year)),
  by = "day"
)
my_lat <- 33.7490
my_lon <- -84.3880
sunset_times <- getSunlightTimes(
  date = my_dates,
  lat = my_lat,
  lon = my_lon,
  keep = "sunset",
  tz = "America/New_York"
)
sunset_times$doy <- yday(sunset_times$date)
sunset_times$mod_sunset <- hour(sunset_times$sunset) * 60 + minute(sunset_times$sunset) # nolint
sunset_times <- sunset_times |> select(doy, mod_sunset)

# filter otm data
otm_data_ATL <- otms_data |>
  filter(location == "ATL") |>
  arrange(id, year, doy, mod) |>
  filter(microhabitat %in% c("branch", "leaflitter")) |>
  left_join(sunset_times, by = "doy") |>
  filter(mod > mod_sunset & mod < mod_sunset + 60) |>
  mutate(
    year = year + 2000
  ) |>
  mutate(
    temp = ifelse(year == 2025 & doy < 200, NA, temp),
    temp = ifelse(year == 2026 & doy > 30, NA, temp),
  ) |>
  filter(!is.na(temp)) |>
  group_by(year, doy, microhabitat) |>
  summarise(
    mean_temp = mean(temp, na.rm = TRUE),
    max_temp = max(temp, na.rm = TRUE),
    min_temp = min(temp, na.rm = TRUE),
    .groups = "drop"
  ) |>
  mutate(
    date = as.Date(paste(year, doy), format = "%Y %j")
  )

# predict CTmin at ATL latitude
ctmin_atl <- 57 - 1.68 * my_lat - 41.67 + 1.48 * my_lat

# plot observation probability and OTM data
obs_plot <- ggplot() +
  geom_hline(
    yintercept = c(1, 0, ctmin_atl / max(otm_data_ATL$mean_temp, na.rm = TRUE)),
    linetype = "dashed",

    alpha = 0.5
  ) +
  annotate(

    label = expression(CT[min]),
    color = "steelblue",
    size = 5,
    fill = "white",
    vjust = 1
  ) +
  geom_line(
    data = filter(otm_data_ATL, microhabitat == "branch"),
    aes(x = date, y = mean_temp / max(otm_data_ATL$mean_temp, na.rm = TRUE)),
    color = "forestgreen", linewidth = 1
  ) +
  geom_line(
    data = filter(otm_data_ATL, microhabitat == "leaflitter"),
    aes(x = date, y = mean_temp / max(otm_data_ATL$mean_temp, na.rm = TRUE)),
    color = "darkorange", linewidth = 1
  ) +
  geom_point(
    data = survey_sum,
    aes(x = date, y = presence),
    color = "darkgray", size = 3, alpha = 0.75
  ) +
  geom_smooth(

otms_data_sum |>
  group_by(location, lat, microhabitat) |>
  summarise(
    mean_min_temp = mean(min_temp, na.rm = TRUE),
    sd_min_temp = sd(min_temp, na.rm = TRUE),
    n = n()
  ) |>
  mutate(
    est = paste(
      round(mean_min_temp, 1), "±",
      round(sd_min_temp, 1), "°C (n =", n, ")"
    ),
  ) |>
  mutate(
    microhabitat = ifelse(microhabitat == "branch", "Canopy", "Leaf litter")
  ) |>
  select(location, lat, microhabitat, est, mean_min_temp) |>
  pivot_wider(
    names_from = microhabitat,
    values_from = c(est, mean_min_temp)
  ) |>
  mutate(
    buffering = round(`mean_min_temp_Leaf litter` - mean_min_temp_Canopy, 1)
  ) |>
  select(location, lat, est_Canopy, `est_Leaf litter`, buffering) |>
  kbl(
    col.names = c("Location", "Latitude (°N)", "Canopy", "Leaf litter", "Average Buffering (°C)"),
    align = "c",
  ) |>
  kable_styling(
    full_width = FALSE,
    position = "center",
    font_size = 12
  )
