Overview

Acquire Data

The first step is to acquire the data needed for the analysis. This includes:

  • Geographic ranges of lizards which we obtain from the GARD database.

  • Global minimum annual temperature data which we obtain from WorldClim.

  • Global elevation data which we obtain from the elevatr package.

Code 
# read in lizard range data
range_shapefile <- st_read(
  "raw_data/lizard_ranges/doi_10_5061_dryad_9cnp5hqmb__v20220427"
)
ranges_vector <- terra::vect(range_shapefile)

# get WorldClim minimum annual temperature data
global_min_temp <- geodata::worldclim_global(
  var = "tmin",
  res = 2.5,
  path = "raw_data"
)

# find minimum value across all layers of the raster for each cell
global_min_temp <- global_min_temp |> terra::app(fun = min, na.rm = TRUE)

# try WorldClim elevation first; fall back to elevatr if server is unavailable
global_elev <- elevatr::get_elev_raster(
  locations = global_min_temp,
  z = 3,
  prj = terra::crs(global_min_temp),
  clip = "locations"
)
global_elev <- terra::rast(global_elev)
global_elev <- terra::crop(global_elev, global_min_temp)
global_elev <- terra::project(global_elev, global_min_temp)
global_elev <- terra::mask(global_elev, global_min_temp)

# add elevation layer to global_min_temp raster
global_min_temp <- c(global_min_temp, global_elev)

# save datasets
saveRDS(global_min_temp, "data/global_min_temp.rds")
saveRDS(ranges_vector, "data/ranges_vector.rds")

Analysis

Next, for each species we will extract the minimum annual temperature data across its geographic range and determine:

  • The proportion and area of the species range that is below its CTmin.

  • The maximum absolute latitude within the species range at which the minimum annual temperature is greater than or equal to its CTmin.

Code 
# read in CTmin data
global_min_temp <- readRDS("data/global_min_temp.rds")
ranges_vector <- readRDS("data/ranges_vector.rds")

# filter CTmin data to only species within the range shapefile
ctmin_data <- ctmin_data |>
  filter(species %in% ranges_vector$binomial) |>
  filter(!is.na(min_elev) & !is.na(max_elev))

# define function to convert latitude/longitude degrees to cell area in km^2
deg_to_area <- function(lat_deg, delta_lat = 0.0416, delta_long = 0.0416) {

  # Convert latitude from degrees to radians
  lat_rad <- lat_deg * pi / 180

  # WGS84 ellipsoid constants
  a <- 6378.1370      # Semi-major axis (equatorial radius in km)
  b <- 6356.7523142   # Semi-minor axis (polar radius in km)

  # Calculate the radius of curvature in the meridian (North-South)
  e_sq <- (a^2 - b^2) / a^2
  denominator <- (1 - e_sq * sin(lat_rad)^2)^(1.5)
  m <- (a * (1 - e_sq)) / denominator

  # Length of 1 degree of latitude in km at this specific latitude
  km_per_degree_lat <- m * (pi / 180)
  lat_dist <- km_per_degree_lat * delta_lat

  # WGS84 Earth semi-major axis in km
  a_km <- 6378.137

  # Flattening factor
  f <- 1 / 298.257223563

  # Calculate distance of 1 degree longitude along the parallel
  dist_km <- (pi / 180) * a_km * cos(lat_rad) / sqrt(1 - (2 * f - f^2) * sin(lat_rad)^2) # nolint
  lon_dist <- delta_long * dist_km

  # Return cell area in km^2
  lat_dist * lon_dist
}

# loop to extract data
results_list <- vector("list", nrow(ctmin_data))

# holder list for latitudinal change in min temperature across species range
lat_temp_change_list <- list()

for (i in seq_along(ctmin_data$species)) {

  tryCatch({

    # filter data for species of interest
    species_name <- ctmin_data$species[i]
    species_range <- ranges_vector[ranges_vector$binomial == species_name, ]
    species_ctmin <- ctmin_data$ctmin[ctmin_data$species == species_name][1]

    # get minimum and maximum latitude within range
    min_lat <- min(abs(terra::ext(species_range)[3:4]))
    max_lat <- max(abs(terra::ext(species_range)[3:4]))

    # get proportion of range below CTmin
    species_min_temp <- terra::extract(
      global_min_temp, species_range, xy = TRUE
    )
    colnames(species_min_temp) <- c("id", "tmin", "elev", "x", "y")

    # filter elevations outside of range
    species_min_temp <- species_min_temp |>
      filter(elev >= ctmin_data$min_elev[i] & elev <= ctmin_data$max_elev[i])

    # calculate area of each cell based on latitude and that all cells re
    species_min_temp$area_km2 <- deg_to_area(species_min_temp$y)

    # get proportion of cells below CTmin
    props <- species_min_temp |>
      summarise(
        total_area = sum(area_km2, na.rm = TRUE),
        area_below_ctmin = sum(area_km2[tmin < species_ctmin], na.rm = TRUE),
      ) |>
      mutate(
        prop_below_ctmin = area_below_ctmin / total_area
      )
    prop_below_ctmin <- props |> pull(prop_below_ctmin)
    area_below_ctmin <- props |> pull(area_below_ctmin)
    total_area <- props |> pull(total_area)

    # get maximum absolute latitude within range at which Tmin is >= CTmin
    above_ctmin <- species_min_temp |> filter(tmin >= species_ctmin)
    max_lat_above_ctmin <- if (nrow(above_ctmin) == 0) {
      NA_real_
    } else {
      max(abs(above_ctmin$y), na.rm = TRUE)
    }

    # store results in pre-allocated list
    results_list[[i]] <- tibble(
      species = species_name,
      ctmin = species_ctmin,
      min_lat = min_lat,
      max_lat = max_lat,
      prop_below_ctmin = prop_below_ctmin,
      area_below_ctmin = area_below_ctmin,
      total_area = total_area,
      max_lat_above_ctmin = max_lat_above_ctmin
    )

    # get latitudinal change in min temperature across species range
    lat_temp_change_list[[i]] <- species_min_temp |>
      mutate(abs_lat = abs(round(y, 1))) |>
      group_by(abs_lat) |>
      summarise(
        mean_min_temp = mean(tmin, na.rm = TRUE),
        max_min_temp = max(tmin, na.rm = TRUE),
        min_min_temp = min(tmin, na.rm = TRUE)
      ) |>
      ungroup() |>
      mutate(species = species_name)

  }, error = function(e) {
    message(
      "Error on species ", i, " (", ctmin_data$species[i], "): ", e$message
    )
  })

  # clear up memory
  rm(species_range, species_min_temp, above_ctmin)
  terra::tmpFiles(remove = TRUE)
  gc()

  print(paste0("Processed species ", i, " of ", nrow(ctmin_data)))

}

# bind rows in results list
results <- bind_rows(results_list)

# bind all species latitudinal temperature change data into a single data frame
lat_temp_change_df <- bind_rows(lat_temp_change_list)

# save datasets
saveRDS(results, "data/ctmin_results.rds")
saveRDS(lat_temp_change_df, "data/lat_temp_change_df.rds")
