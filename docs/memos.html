microclimate_temps <- readRDS("data/microclimate_temps.rds")

# define temperature thresholds
lethal <- 0
ctmin <- 5.4
retreat <- 6.6

# define thresholds
thresholds <- microclimate_temps |>
  filter(z == 1.8) |>
  mutate(
    thresh = case_when(
      min_annual_tb < lethal ~ "lethal",
      min_annual_tb >= lethal & min_annual_tb < ctmin ~ "ctmin",
      min_annual_tb >= ctmin & min_annual_tb < retreat ~ "retreat",
      TRUE ~ "rest"
    )
  ) |>
  select(lon, lat, thresh)

# merge microclimate temps with thresholds
microclimate_temps <- merge(
  microclimate_temps,
  thresholds,
  by = c("lon", "lat"),
  all.x = TRUE
)

# define conditions to test
threshold_names <- unique(microclimate_temps$thresh)
veg <- seq(0, 1, by = 0.01)
heights <- unique(microclimate_temps$z)

# create grid to store values
tss_grid <- expand.grid(
  thresh = threshold_names,
  veg = veg,
  z = heights
)
tss_grid$tss <- NA_real_

# loop through thresholds, tree covers, and heights to calculate TSS
for (i in seq_len(nrow(tss_grid))) {

  # add progress tracker
  cat("Processing row", i, "of", nrow(tss_grid), "\n")

  if (tss_grid$thresh[i] == "lethal") {
    data_subset <- microclimate_temps |>
      filter(
        thresh == "lethal",
        z == tss_grid$z[i]
      )
  } else if (tss_grid$thresh[i] == "ctmin") {
    data_subset <- microclimate_temps |>
      filter(
        thresh %in% c("lethal", "ctmin"),
        z == tss_grid$z[i]
      )
  } else if (tss_grid$thresh[i] == "retreat") {
    data_subset <- microclimate_temps |>
      filter(
        thresh != "rest",
        z == tss_grid$z[i]
      )
  } else {
    data_subset <- microclimate_temps |>
      filter(
        z == tss_grid$z[i]
      )
  }

  tss_value <- data_subset |>
    mutate(
      pred = min_annual_tb > lethal & veg > tss_grid$veg[i]
    ) |>
    summarise(
      tp = sum(pred & presence),
      fp = sum(pred & !presence),
      fn = sum(!pred & presence),
      tn = sum(!pred & !presence)
    ) |>
    mutate(tss = (tp / (tp + fn)) + (tn / (tn + fp)) - 1) |>
    pull(tss)
  tss_grid$tss[i] <- tss_value

}

# save tss grid
saveRDS(tss_grid, "data/tss_grid.rds")

ggplot() +
  geom_tile(
    data = tss_grid,
    aes(x = veg, y = as.factor(z), fill = tss)
  ) +
  geom_point(
    data = tss_grid |>
      group_by(thresh) |>
      filter(tss == max(tss, na.rm = TRUE)) |>
      ungroup(),

    shape = 1
